cmake_minimum_required(VERSION 3.20)
project(deadspace_cpu_probe LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DYNARMIC_FRONTENDS A32 CACHE STRING "" FORCE)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_NAME STREQUAL "Generic")
  set(FMT_OS OFF CACHE BOOL "No POSIX file wrapper on Horizon" FORCE)
  set(DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT ON CACHE BOOL "Use libnx JIT transitions" FORCE)
endif()
set(DYNARMIC_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/dynarmic" CACHE PATH "Pinned Dynarmic source")
add_subdirectory("${DYNARMIC_SOURCE_DIR}" dynarmic)
add_library(ds_cpu_probe STATIC hd_texture.cpp ../../third_party/powervr/PVRTDecompress.cpp guest_gl.cpp ../../src/atc_decompress.cpp guest_jni.cpp guest_image.cpp cpu_probe.cpp constructor_probe.cpp lifecycle_probe.cpp ../source/preflight.cpp)
target_include_directories(ds_cpu_probe PUBLIC . ../include)
target_link_libraries(ds_cpu_probe PUBLIC dynarmic)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Generic")
  add_executable(ds-hd-test hd_texture.cpp ../../src/atc_decompress.cpp ../../third_party/powervr/PVRTDecompress.cpp ../tests/hd_texture_test.cpp)
  target_include_directories(ds-hd-test PRIVATE .)
  add_executable(ds-native-gl-test hd_texture.cpp ../../third_party/powervr/PVRTDecompress.cpp guest_gl.cpp ../../src/atc_decompress.cpp ../tests/guest_gl_native_test.cpp)
  target_include_directories(ds-native-gl-test PRIVATE .)
  target_link_libraries(ds-native-gl-test PRIVATE ${CMAKE_DL_LIBS})
  add_executable(ds-memory-test ../tests/guest_memory_test.cpp)
  target_include_directories(ds-memory-test PRIVATE .)
  add_executable(ds-jni-test guest_jni.cpp ../tests/guest_jni_test.cpp)
  target_include_directories(ds-jni-test PRIVATE .)
  add_executable(ds-cpu-probe host_main.cpp)
  target_link_libraries(ds-cpu-probe PRIVATE ds_cpu_probe)
  add_executable(ds-loader-test ../tests/runtime_loader_test.cpp)
  target_link_libraries(ds-loader-test PRIVATE ds_cpu_probe)
endif()
